**Step 2: Create the verify-oracle-providers.ts script**

Create `scripts/verify-oracle-providers.ts`. This script:
1. Reads on-chain `oracleProviderForToken` for all 7 configured tokens
2. Compares each against `config.pythLazerFeedProviderAddress`
3. Reports matches and mismatches in a clear table
4. When run with `--fix` flag, calls `DataStore.setAddress()` to update mismatched tokens

The script should use the same viem client infrastructure as the keeper (imported from `src/core/blockchain/client.ts`). It should work standalone via `pnpm tsx scripts/verify-oracle-providers.ts`.

```typescript
#!/usr/bin/env node
/**
 * Verify and fix on-chain oracleProviderForToken mappings in DataStore.
 *
 * Usage:
 *   pnpm tsx scripts/verify-oracle-providers.ts          # Verify only (read-only)
 *   pnpm tsx scripts/verify-oracle-providers.ts --fix     # Fix mismatches (writes to chain)
 *
 * Requires .env to be configured with:
 *   RPC_URL, PRIVATE_KEY, DATA_STORE_ADDRESS, PYTH_LAZER_FEED_PROVIDER_ADDRESS
 */

import { config } from "../src/config.js";
import { TOKEN_ADDRESSES, PYTH_LAZER_FEED_CONFIGS } from "../src/config/tokens.js";
import { getPublicClient, getWalletClient, getAccount } from "../src/core/blockchain/client.js";
import { dataStoreAbi } from "../src/core/blockchain/contracts/abis/dataStore.js";
import { keccak256, encodeAbiParameters, type Address } from "viem";

const FIX_MODE = process.argv.includes("--fix");

// Build the ORACLE_PROVIDER_FOR_TOKEN base key (matches Solidity abi.encode)
const ORACLE_PROVIDER_FOR_TOKEN_BASE = keccak256(
  encodeAbiParameters([{ type: "string" }], ["ORACLE_PROVIDER_FOR_TOKEN"])
);

function oracleProviderForTokenKey(token: Address): `0x${string}` {
  return keccak256(
    encodeAbiParameters(
      [{ type: "bytes32" }, { type: "address" }],
      [ORACLE_PROVIDER_FOR_TOKEN_BASE, token]
    )
  );
}

// Reverse map: address -> symbol for display
const addressToSymbol: Record<string, string> = {};
for (const [symbol, addr] of Object.entries(TOKEN_ADDRESSES)) {
  addressToSymbol[addr.toLowerCase()] = symbol;
}

async function main() {
  console.log("=== Oracle Provider Verification ===\n");
  console.log(`DataStore:            ${config.dataStoreAddress}`);
  console.log(`Expected Provider:    ${config.pythLazerFeedProviderAddress}`);
  console.log(`Mode:                 ${FIX_MODE ? "VERIFY + FIX" : "VERIFY ONLY"}\n`);

  const publicClient = getPublicClient();

  // Get all configured tokens (from PYTH_LAZER_FEED_CONFIGS)
  const allTokens = PYTH_LAZER_FEED_CONFIGS.map(f => f.token);

  const matches: { symbol: string; token: Address }[] = [];
  const mismatches: { symbol: string; token: Address; onChain: Address }[] = [];
  const errors: { symbol: string; token: Address; error: string }[] = [];

  for (const token of allTokens) {
    const symbol = addressToSymbol[token.toLowerCase()] || "UNKNOWN";
    const key = oracleProviderForTokenKey(token);

    try {
      const onChainProvider = await publicClient.readContract({
        address: config.dataStoreAddress,
        abi: dataStoreAbi,
        functionName: "getAddress",
        args: [key],
      }) as Address;

      if (onChainProvider.toLowerCase() === config.pythLazerFeedProviderAddress.toLowerCase()) {
        matches.push({ symbol, token });
        console.log(`  OK   ${symbol.padEnd(6)} ${token} -> ${onChainProvider}`);
      } else {
        mismatches.push({ symbol, token, onChain: onChainProvider });
        console.log(`  MISS ${symbol.padEnd(6)} ${token} -> ${onChainProvider} (expected: ${config.pythLazerFeedProviderAddress})`);
      }
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      errors.push({ symbol, token, error: msg });
      console.log(`  ERR  ${symbol.padEnd(6)} ${token} -> ${msg.slice(0, 80)}`);
    }
  }

  console.log(`\n--- Summary ---`);
  console.log(`Matches:    ${matches.length}/${allTokens.length}`);
  console.log(`Mismatches: ${mismatches.length}/${allTokens.length}`);
  if (errors.length > 0) {
    console.log(`Errors:     ${errors.length}/${allTokens.length}`);
  }

  if (mismatches.length === 0) {
    console.log("\nAll tokens have correct oracle providers. No action needed.");
    process.exit(0);
  }

  if (!FIX_MODE) {
    console.log("\nTo fix mismatches, re-run with --fix flag:");
    console.log("  pnpm tsx scripts/verify-oracle-providers.ts --fix");
    process.exit(1);
  }

  // Fix mode: update mismatched tokens
  console.log("\n=== Fixing Mismatches ===\n");

  const walletClient = getWalletClient();
  const account = getAccount();

  for (const { symbol, token } of mismatches) {
    const key = oracleProviderForTokenKey(token);

    try {
      console.log(`Setting oracleProviderForToken for ${symbol} (${token})...`);
      const hash = await walletClient.writeContract({
        address: config.dataStoreAddress,
        abi: dataStoreAbi,
        functionName: "setAddress",
        args: [key, config.pythLazerFeedProviderAddress],
        account,
      } as any);

      console.log(`  TX sent: ${hash}`);
      const receipt = await publicClient.waitForTransactionReceipt({
        hash,
        timeout: 30_000,
      });

      if (receipt.status === "reverted") {
        console.log(`  REVERTED — the keeper wallet may not have CONTROLLER role on DataStore`);
        console.log(`  Run configureOracleTokens.ts from the contracts repo instead:`);
        console.log(`    cd /Users/ken/Projects/0xM/0xmarkets_contract`);
        console.log(`    npx hardhat deploy --tags ConfigureOracleTokens --network baseSepolia`);
      } else {
        console.log(`  Confirmed in block ${receipt.blockNumber}`);
      }
    } catch (err) {
      const msg = err instanceof Error ? err.message : String(err);
      console.log(`  FAILED: ${msg.slice(0, 120)}`);
      if (msg.includes("reverted") || msg.includes("unauthorized") || msg.includes("Unauthorized")) {
        console.log(`\n  The keeper wallet likely lacks CONTROLLER role on DataStore.`);
        console.log(`  Use the contracts repo deployer account instead:`);
        console.log(`    cd /Users/ken/Projects/0xM/0xmarkets_contract`);
        console.log(`    npx hardhat deploy --tags ConfigureOracleTokens --network baseSepolia\n`);
        break;
      }
    }
  }

  // Re-verify after fixes
  console.log("\n=== Re-verification ===\n");
  let allFixed = true;
  for (const { symbol, token } of mismatches) {
    const key = oracleProviderForTokenKey(token);
    try {
      const onChainProvider = await publicClient.readContract({
        address: config.dataStoreAddress,
        abi: dataStoreAbi,
        functionName: "getAddress",
        args: [key],
      }) as Address;

      if (onChainProvider.toLowerCase() === config.pythLazerFeedProviderAddress.toLowerCase()) {
        console.log(`  FIXED ${symbol.padEnd(6)} ${token} -> ${onChainProvider}`);
      } else {
        console.log(`  STILL WRONG ${symbol.padEnd(6)} ${token} -> ${onChainProvider}`);
        allFixed = false;
      }
    } catch (err) {
      console.log(`  ERR   ${symbol.padEnd(6)} ${token}`);
      allFixed = false;
    }
  }

  process.exit(allFixed ? 0 : 1);
}

main().catch((err) => {
  console.error("Fatal:", err);
  process.exit(1);
});
```
