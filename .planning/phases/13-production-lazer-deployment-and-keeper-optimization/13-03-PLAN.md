<objective>
Add per-token oracle routing so that `buildOracleParams()` selects the correct oracle provider per-token based on startup entitlement verification results.

Purpose: ORCL-02 requires the keeper to route crypto tokens (WETH, WBTC, USDC) to Lazer and FX tokens (EUR, GBP, GOLD, JPY) to Hermes/Chainlink when Lazer entitlements are missing. Currently all tokens are routed uniformly via the global `config.oracleMode`. Since all 7 Lazer feeds ARE currently receiving data, the primary path routes everything through Lazer, but the per-token capability ensures graceful degradation if individual feeds fail.

Output: Modified keeper that stores entitlement state and routes oracle params per-token.
</objective>

```typescript
// Per-token Lazer entitlement state — set at startup by verifyLazerFeeds()
// Tokens in this set have confirmed Lazer WebSocket data.
// Tokens NOT in this set should fall back to Hermes for oracle params.
let lazerEntitledTokens: Set<string> = new Set();

export function setLazerEntitledTokens(tokens: Address[]): void {
  lazerEntitledTokens = new Set(tokens.map(t => t.toLowerCase()));
}

export function getLazerEntitledTokens(): Set<string> {
  return lazerEntitledTokens;
}

export function isTokenLazerEntitled(token: Address): boolean {
  return lazerEntitledTokens.has(token.toLowerCase());
}
```

```typescript
const tokensToPrice = tokens;

// Partition tokens into Lazer-entitled and Hermes-fallback groups
const lazerTokens: Address[] = [];
const hermesTokens: Address[] = [];

if (config.oracleMode === "lazer" || config.oracleMode === "both") {
  for (const token of tokensToPrice) {
    if (isTokenLazerEntitled(token) && getPythLazerOracle()?.hasFeed(token)) {
      lazerTokens.push(token);
    } else {
      hermesTokens.push(token);
    }
  }
} else {
  // Pure hermes mode — all tokens go to Hermes
  hermesTokens.push(...tokensToPrice);
}

// Update Lazer prices on-chain for entitled tokens
if (lazerTokens.length > 0) {
  const pythLazerOracle = getPythLazerOracle();
  if (pythLazerOracle) {
    for (const token of lazerTokens) {
      const isFresh = await this.isStoredPriceFresh(token);
      if (isFresh) {
        log.info({ token }, "stored price is fresh, skipping updatePriceOnChain");
        continue;
      }

      log.info({ token }, "stored price stale or missing, updating synchronously");
      try {
        const txHash = await pythLazerOracle.updatePriceOnChain(token);
        if (!txHash && hermesTokens.length === 0) {
          // Only throw if there's no Hermes fallback available
          throw new Error(
            `No Pyth Lazer update available for token ${token} — WebSocket may not be connected yet`
          );
        }
        if (!txHash) {
          // Lazer update failed but Hermes is available — move token to Hermes group
          log.warn({ token }, "Lazer update failed, falling back to Hermes for this token");
          hermesTokens.push(token);
          lazerTokens.splice(lazerTokens.indexOf(token), 1);
        }
      } catch (error) {
        const errMsg = error instanceof Error ? error.message : String(error);
        if (errMsg.includes("disconnected") || errMsg.includes("cache stale") || errMsg.includes("WebSocket")) {
          log.error({ token }, "oracle price update failed — WebSocket disconnected and cache stale");
        }
        log.error({ err: error, token }, "failed to update Pyth Lazer price");
        if (config.oracleMode === "lazer" && hermesTokens.length === 0) {
          throw error;
        }
        // Fall back to Hermes for this token
        log.warn({ token }, "Lazer update failed, falling back to Hermes for this token");
        hermesTokens.push(token);
        // Remove from lazerTokens to avoid including it in Lazer params
        const idx = lazerTokens.indexOf(token);
        if (idx >= 0) lazerTokens.splice(idx, 1);
      }
    }
  }
}

// Build combined oracle params
const resultTokens: Address[] = [];
const resultProviders: Address[] = [];
const resultData: Hex[] = [];

// Add Lazer params (stored on-chain, empty data)
const pythLazerProvider = config.pythLazerFeedProviderAddress as Address;
for (const token of lazerTokens) {
  resultTokens.push(token);
  resultProviders.push(pythLazerProvider);
  resultData.push("0x" as Hex);
}

// Add Hermes params (fetched via REST API)
if (hermesTokens.length > 0) {
  const hermesWithFeeds = hermesTokens.filter(token => pythOracle.hasFeed(token));
  if (hermesWithFeeds.length > 0) {
    log.info({ tokenCount: hermesWithFeeds.length }, "fetching Pyth Hermes prices for fallback tokens");
    const priceData = await pythOracle.fetchPrices(hermesWithFeeds);
    const hermesParams = pythOracle.buildSetPricesParams(priceData);
    resultTokens.push(...hermesParams.tokens);
    resultProviders.push(...hermesParams.providers);
    resultData.push(...hermesParams.data);
  } else if (lazerTokens.length === 0) {
    // No tokens priced by either provider
    log.warn("no oracle feeds configured for any tokens, using empty params");
    return { tokens: [], providers: [], data: [] };
  }
}

log.info(
  { lazerCount: lazerTokens.length, hermesCount: hermesTokens.length, total: resultTokens.length },
  "built per-token oracle params"
);
return { tokens: resultTokens, providers: resultProviders, data: resultData };
```
