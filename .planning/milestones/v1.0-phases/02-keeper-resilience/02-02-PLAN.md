Add a new public method `cancelExpiredDeposits()`:
```typescript
async cancelExpiredDeposits(): Promise<number> {
  const prisma = await this.getPrisma();
  const pendingDeposits = await prisma.depositRequest.findMany({
    where: { status: "PENDING" },
    select: { requestKey: true },
  });

  if (pendingDeposits.length === 0) return 0;

  // Read REQUEST_EXPIRATION_TIME from DataStore (seconds as bigint)
  const requestExpirationTime = await this.dataStore.getUint(REQUEST_EXPIRATION_TIME_KEY);
  if (requestExpirationTime === 0n) {
    // Expiry is disabled on-chain — nothing to cancel
    return 0;
  }

  const nowSeconds = BigInt(Math.floor(Date.now() / 1000));
  const walletClient = getWalletClient();
  const publicClient = getPublicClient();
  const account = getAccount();
  let cancelled = 0;

  for (const { requestKey } of pendingDeposits) {
    const deposit = await this.reader.getDeposit(requestKey as Hex);
    if (!deposit) {
      // Already removed from chain — cleanupStaleDeposits will handle
      continue;
    }

    // Check if deposit struct is zeroed (already consumed)
    if (!deposit.addresses?.account || deposit.addresses.account === ZERO_ADDRESS) {
      continue;
    }

    const expiredAt = deposit.numbers.updatedAtTime + requestExpirationTime;
    if (nowSeconds <= expiredAt) continue; // not yet expired

    console.log(
      `[DepositScanner] Deposit ${requestKey} expired (updatedAt=${deposit.numbers.updatedAtTime}, ` +
      `expiration=${requestExpirationTime}, now=${nowSeconds}), cancelling on-chain`
    );

    try {
      const txHash = await walletClient.writeContract({
        address: config.depositHandlerAddress,
        abi: depositHandlerAbi,
        functionName: "cancelDeposit",
        args: [requestKey as Hex],
        account,
      });
      await publicClient.waitForTransactionReceipt({ hash: txHash, timeout: 60_000 });
      await prisma.depositRequest.update({
        where: { requestKey },
        data: {
          status: "CANCELLED",
          errorReason: "Deposit expired — cancelled on-chain by keeper",
        },
      });
      cancelled++;
      console.log(`[DepositScanner] Cancelled expired deposit ${requestKey}: txHash=${txHash}`);
    } catch (error) {
      console.error(`[DepositScanner] Failed to cancel expired deposit ${requestKey}:`, error);
      // Don't throw — try to cancel remaining deposits
    }
  }

  return cancelled;
}
```

Also add LIFE-03 documentation comment at the top of the `scan()` method:
```typescript
/**
 * Scan for new deposit requests on-chain and return all pending keys for execution.
 *
 * LIFE-03: Restart recovery is handled by design. On restart, this method:
 * 1. Reads DEPOSIT_LIST from DataStore (includes deposits created during downtime)
 * 2. Filters for keys not yet in DB -> stores them as PENDING
 * 3. Returns ALL PENDING keys (both pre-existing and newly discovered)
 *
 * There is no "EXECUTING" intermediate state. Deposits in-flight during a crash
 * remain PENDING in the DB and will be retried. Idempotency is guaranteed by the
 * on-chain EmptyDeposit() guard — if the deposit was already executed, the keeper
 * detects the zeroed struct and marks it CANCELLED (not FAILED).
 */
```

Create a standalone viem script that grants CONTROLLER role to the keeper wallet. This must be run with the deployer's private key (which has ROLE_ADMIN).

```javascript
/**
 * Grant CONTROLLER role to the keeper wallet on Base Sepolia.
 *
 * Required: The deployer wallet (which has ROLE_ADMIN) must sign.
 * Usage: DEPLOYER_KEY=0x... node scripts/grant-keeper-controller-role.mjs
 *
 * This is a ONE-TIME operation. After running, the keeper can call
 * DepositHandler.cancelDeposit() to cancel expired deposits.
 */
import { createPublicClient, createWalletClient, http, keccak256, encodePacked } from 'viem';
import { baseSepolia } from 'viem/chains';
import { privateKeyToAccount } from 'viem/accounts';

// Configuration — update these if contract addresses change
const ROLE_STORE_ADDRESS = '0x...'; // TODO: Fill from deployment
const KEEPER_ADDRESS = '0x48Cb0d738C9B3F44F60f7338F788fa093FD25828';
const RPC_URL = 'https://sepolia.base.org';

const CONTROLLER_ROLE = keccak256(encodePacked(['string'], ['CONTROLLER']));

const roleStoreAbi = [
  {
    type: 'function',
    name: 'grantRole',
    inputs: [
      { name: 'account', type: 'address' },
      { name: 'roleKey', type: 'bytes32' },
    ],
    outputs: [],
    stateMutability: 'nonpayable',
  },
  {
    type: 'function',
    name: 'hasRole',
    inputs: [
      { name: 'account', type: 'address' },
      { name: 'roleKey', type: 'bytes32' },
    ],
    outputs: [{ name: '', type: 'bool' }],
    stateMutability: 'view',
  },
];

async function main() {
  const deployerKey = process.env.DEPLOYER_KEY;
  if (!deployerKey) {
    console.error('ERROR: Set DEPLOYER_KEY environment variable (deployer private key with ROLE_ADMIN)');
    process.exit(1);
  }

  const account = privateKeyToAccount(deployerKey.startsWith('0x') ? deployerKey : `0x${deployerKey}`);
  const publicClient = createPublicClient({ chain: baseSepolia, transport: http(RPC_URL) });
  const walletClient = createWalletClient({ chain: baseSepolia, transport: http(RPC_URL), account });

  console.log(`Deployer: ${account.address}`);
  console.log(`Keeper:   ${KEEPER_ADDRESS}`);
  console.log(`Role:     CONTROLLER (${CONTROLLER_ROLE})`);

  // Check if already granted
  const hasRole = await publicClient.readContract({
    address: ROLE_STORE_ADDRESS,
    abi: roleStoreAbi,
    functionName: 'hasRole',
    args: [KEEPER_ADDRESS, CONTROLLER_ROLE],
  });

  if (hasRole) {
    console.log('Keeper already has CONTROLLER role. No action needed.');
    return;
  }

  console.log('Granting CONTROLLER role to keeper...');
  const txHash = await walletClient.writeContract({
    address: ROLE_STORE_ADDRESS,
    abi: roleStoreAbi,
    functionName: 'grantRole',
    args: [KEEPER_ADDRESS, CONTROLLER_ROLE],
  });

  console.log(`Transaction submitted: ${txHash}`);
  const receipt = await publicClient.waitForTransactionReceipt({ hash: txHash });
  console.log(`Transaction confirmed in block ${receipt.blockNumber}`);

  // Verify
  const verified = await publicClient.readContract({
    address: ROLE_STORE_ADDRESS,
    abi: roleStoreAbi,
    functionName: 'hasRole',
    args: [KEEPER_ADDRESS, CONTROLLER_ROLE],
  });
  console.log(`Verified: keeper has CONTROLLER = ${verified}`);
}

main().catch(err => { console.error(err); process.exit(1); });
```
