The `/metrics` endpoint should return:

```typescript
app.get("/metrics", (_req, res) => {
  const now = Date.now();
  const nowSec = BigInt(Math.floor(now / 1000));

  // Queue stats
  const queueStats = getQueueStats();

  // Feed freshness per token
  const oracleService = getPythLazerOracle();
  const feedFreshness: Record<string, { feedId: number; ageSeconds: number | null; hasData: boolean }> = {};
  if (oracleService) {
    for (const feedConfig of PYTH_LAZER_FEED_CONFIGS) {
      const update = oracleService.getLatestUpdate(feedConfig.token);
      feedFreshness[feedConfig.token] = {
        feedId: feedConfig.feedId,
        ageSeconds: update ? Number(nowSec - update.timestamp) : null,
        hasData: !!update,
      };
    }
  }

  // Execution rates (from healthState)
  const uptime = now - healthState.startedAt.getTime();
  const uptimeHours = uptime / (1000 * 60 * 60);
  const totalExecutions =
    healthState.executionCounts.deposits +
    healthState.executionCounts.withdrawals +
    healthState.executionCounts.orders;

  res.json({
    timestamp: new Date().toISOString(),
    uptime: Math.floor(uptime / 1000),
    queue: queueStats,
    feeds: feedFreshness,
    executions: {
      counts: healthState.executionCounts,
      total: totalExecutions,
      ratePerHour: uptimeHours > 0 ? +(totalExecutions / uptimeHours).toFixed(2) : 0,
    },
    latency: latencyTracker.getPercentiles(),
    oracleMode: config.oracleMode,
    oracleConnected: healthState.oracleConnected,
    wsConnected: healthState.wsConnected,
  });
});
```

```dockerfile
FROM base AS production
RUN apt-get update -y && apt-get install -y curl && rm -rf /var/lib/apt/lists/*

COPY --from=build /app/dist ./dist
COPY --from=build /app/prisma ./prisma
```
