```typescript
import { keccak256, encodeAbiParameters, parseAbi } from "viem";

export async function verifyOracleProviderConsistency(
  dataStoreAddress: Address,
  tokens: Address[],
  expectedProvider: Address,
  pinoLogger?: Logger
): Promise<{ matches: Address[]; mismatches: { token: Address; onChain: Address }[] }> {
  const moduleLog = pinoLogger ?? log;
  const matches: Address[] = [];
  const mismatches: { token: Address; onChain: Address }[] = [];
  const publicClient = getPublicClient();

  // Build the ORACLE_PROVIDER_FOR_TOKEN base key
  // Matches Solidity: keccak256(abi.encode("ORACLE_PROVIDER_FOR_TOKEN"))
  const baseKey = keccak256(
    encodeAbiParameters([{ type: "string" }], ["ORACLE_PROVIDER_FOR_TOKEN"])
  );

  for (const token of tokens) {
    try {
      const key = keccak256(
        encodeAbiParameters(
          [{ type: "bytes32" }, { type: "address" }],
          [baseKey, token]
        )
      );

      const onChainProvider = await publicClient.readContract({
        address: dataStoreAddress,
        abi: parseAbi(["function getAddress(bytes32 key) view returns (address)"]),
        functionName: "getAddress",
        args: [key],
      });

      if (onChainProvider.toLowerCase() !== expectedProvider.toLowerCase()) {
        mismatches.push({ token, onChain: onChainProvider as Address });
      } else {
        matches.push(token);
      }
    } catch (err) {
      moduleLog.warn({ err, token }, "failed to read oracle provider for token");
    }
  }

  return { matches, mismatches };
}
```

Note: `keccak256`, `encodeAbiParameters`, and `parseAbi` must be added to the existing viem import at the top of the file. The existing import is `import { Address, Hex } from "viem";` — extend it.
